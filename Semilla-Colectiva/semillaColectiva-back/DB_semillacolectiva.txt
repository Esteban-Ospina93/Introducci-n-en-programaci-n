use db_semillacolectiva;

CREATE TABLE usuarios (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    correo VARCHAR(100) UNIQUE NOT NULL,
    contrasena VARCHAR(255) NOT NULL,
    rol ENUM('ADMIN','VOLUNTARIO','INSTITUCION','PROFESIONAL') DEFAULT 'VOLUNTARIO',
    fecha_registro DATETIME DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE voluntarios (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    apellidos VARCHAR(100) UNIQUE,
    usuario_id INT,
    FOREIGN KEY (usuario_id) REFERENCES usuarios(id)
);

CREATE TABLE zonas (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    descripcion TEXT,
    latitud DECIMAL(10, 6),
    longitud DECIMAL(10, 6)
);

CREATE TABLE arboles (
    id INT AUTO_INCREMENT PRIMARY KEY,
    especie VARCHAR(100) NOT NULL,
    fecha_plantacion DATE NOT NULL,
    etapa ENUM('germinacion', 'crecimiento', 'madurez', 'senescencia') DEFAULT 'germinacion',
    estado ENUM('saludable', 'enfermo', 'riesgo', 'transplantado') DEFAULT 'saludable',
    voluntario_id INT,
    zona_id INT,
    FOREIGN KEY (voluntario_id) REFERENCES voluntarios(id),
    FOREIGN KEY (zona_id) REFERENCES zonas(id)
);

CREATE TABLE historial_arboles (
    id INT AUTO_INCREMENT PRIMARY KEY,
    arbol_id INT,
    fecha_cambio DATETIME DEFAULT CURRENT_TIMESTAMP,
    estado_anterior VARCHAR(50),
    estado_nuevo VARCHAR(50),
    usuario_id INT,
    FOREIGN KEY (arbol_id) REFERENCES arboles(id),
    FOREIGN KEY (usuario_id) REFERENCES usuarios(id)
);

CREATE TABLE reportes (
    id INT AUTO_INCREMENT PRIMARY KEY,
    titulo VARCHAR(100),
    descripcion TEXT,
    fecha_generacion DATETIME DEFAULT CURRENT_TIMESTAMP,
    generado_por INT,
    FOREIGN KEY (generado_por) REFERENCES usuarios(id)
);
